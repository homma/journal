#### src/myapp/main.swift
````swift
import SDL3

func main() -> Int32 {

  if !SDL_Init(SDL_INIT_VIDEO) {
    print("SDL_Init failed.")
    return 1
  }

  let w = (
    title: "My SDL3 Window",
    w: Int32(640),
    h: Int32(480),
    flags: UInt64(0)
  )
  let win = SDL_CreateWindow(w.title, w.w, w.h, w.flags)
  if win == nil {
    print("SDL_CreateWindow failed.")
    return 1
  }

  var event = SDL_Event()
  var quit = false

  while quit != true {
    while SDL_PollEvent(&event) {
      if event.type == SDL_EVENT_QUIT.rawValue {
        quit = true
      }
    }
  }

  SDL_DestroyWindow(win)
  SDL_Quit()

  return 0
}

_ = main()
````

#### Package.swift
````swift
// swift-tools-version: 6.3

import PackageDescription

// lib target
let sdl3_path = "sdl3"
let sdl3_target = Target.systemLibrary(name: sdl3_path)

// exec target
let exec_path = "myapp"
let sdl3_dependency = Target.Dependency.target(name: sdl3_path)
let exec_dependencies = [sdl3_dependency]
let exec_target = Target.executableTarget(name: exec_path, dependencies: exec_dependencies)

// targets
let targets = [exec_target, sdl3_target]

// app
let app_name = "myapp"
let app_targets = [exec_path]
let product = Product.executable(name: app_name, targets: app_targets)
let products = [product]

// package
let package = Package(name: app_name, products: products, targets: targets)
````

#### build.sh
````sh
#!/bin/sh -x

SDL3_INC="-Xcc -I$(brew --prefix sdl3)/include/"
SDL3_LIB="-Xlinker -L$(brew --prefix sdl3)/lib"
SDL3_OPT="${SDL3_INC} ${SDL3_LIB}"

swift build -c release ${SDL3_OPT}
````

#### clean.sh
````sh
#!/bin/sh -x

swift package clean
````

#### format.sh
````sh
#!/bin/sh -x

swift format -i -r .
````
