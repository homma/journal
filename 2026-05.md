### D 言語から FLTK を利用する
> date: 2026-05-25  
> topics: d, fltk

`FLTK` の C 言語用のラッパーを作成し、`ImportC` を使用して D 言語のプログラムから利用します。

#### src/fltk.cpp

`FLTK` のラッパー。

````cpp
#include <FL/Fl.H>
#include <FL/Fl_Window.H>

extern "C" {

auto new_fl_window(int left, int top, int width, int height, const char *title)
    -> Fl_Window * {
  auto win = new Fl_Window(left, top, width, height, title);
  return win;
}

void fl_window_show(Fl_Window *win) { win->show(); }

int fl_run() { return Fl::run(); }
}
````

#### src/fltk.h

`FLTK` のラッパーを `ImportC` で使用可能にするためのヘッダー。

````c
typedef struct Fl_Window Fl_Window;

Fl_Window *new_fl_window(int left, int top, int width, int height,
                         const char *title);

void fl_window_show(Fl_Window *win);

int fl_run();
````

#### src/main.d

`FLTK` のウィンドウを作成する D のプログラム。

````d
import std.string : toStringz;

import fltk;

auto main() {

  auto left = 200;
  auto top = 200;
  auto width = 640;
  auto height = 480;
  auto title = "My FLTK Window".toStringz;
  auto win = new_fl_window(left, top, width, height, title);

  fl_window_show(win);

  return fl_run();
}
````

#### dub.sdl

````sdlang
# target name
name "d_fltk_test"

# create an executable
targetType "executable"

# main source files
mainSourceFile "src/main.d"

# objects to link
sourceFiles "src/fltk.o"

# system libraries to link
libs "fltk" "stdc++"

# ldc2 linker flags
lflags "${D_LFLAGS}"

# ImportC search path
cImportPaths "src"
````

#### build.sh
````sh
#!/bin/sh -x

FLTK_INC="-I$(brew --prefix fltk)/include"
FLTK_OPT="${FLTK_INC}"

FLTK_LIB="-L$(brew --prefix fltk)/lib"
export D_LFLAGS=${FLTK_LIB}

c++ src/fltk.cpp -c -o src/fltk.o -std=c++2c ${FLTK_OPT}
dub build
````

#### ビルド
````console
$ chmod +x ./build.sh
$ ./build.sh
````
