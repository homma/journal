### D 言語から wxWidgets を利用する
> date: 2026-05-25  
> topics: d, wxWidgets

`wxWidgets` の C 言語用のラッパーを作成し、`ImportC` を使用して D 言語のプログラムから利用します。

#### src/wx.cpp

`wxWidgets` のラッパー。

````cpp
#include <wx/wx.h>

#include <print>
#include <stdbool.h>

typedef bool (*fp)(void);
static fp wx_init_fn = nullptr;

class MyApp : public wxApp {
public:
  virtual bool OnInit() override;
};

bool MyApp::OnInit() {
  if (!wxApp::OnInit()) {
    std::println("wxApp::OnInit() failed.");
    return false;
  }

  if (!wx_init_fn) {
    std::println("wx_init_fn not set");
    return false;
  }

  return wx_init_fn();
}

wxIMPLEMENT_APP_NO_MAIN(MyApp);

extern "C" {

// run below to generate the header file
// grep '^//| ' src/wx.cpp | sed 's/^\/\/\| //' > src/wx.h

//| #include <stdbool.h>

//| //// wxApp

//| typedef bool (*fp)(void);
//| void wx_main(fp fn);
auto wx_main(fp fn) -> void {
  wx_init_fn = fn;

  int argc = 0;
  wxEntry(argc, (char **)nullptr);
}

//| //// wxFrame
//| typedef struct wxFrame wxFrame;

//| // create
//| wxFrame *new_wx_frame(const char *title);
auto new_wx_frame(const char *title) -> wxFrame * {
  return new wxFrame(nullptr, wxID_ANY, title);
}

//| // show
//| void wx_frame_show(wxFrame *frame);
auto wx_frame_show(wxFrame *frame) -> void { frame->Show(true); }

//| // set menu bar
//| void wx_frame_set_menu_bar(wxFrame *frame, wxMenuBar *menubar);
auto wx_frame_set_menu_bar(wxFrame *frame, wxMenuBar *menu_bar) -> void {
  return frame->SetMenuBar(menu_bar);
}

//| //// wxMenuBar
//| typedef struct wxMenuBar wxMenuBar;

//| // create
//| wxMenuBar *new_wx_menu_bar(void);
auto new_wx_menu_bar(void) -> wxMenuBar * { return new wxMenuBar(); }
}
````

#### src/wx.h

`wxWidgets` のラッパーを `ImportC` で使用可能にするためのヘッダー。

````c
#include <stdbool.h>
//// wxApp
typedef bool (*fp)(void);
void wx_main(fp fn);
//// wxFrame
typedef struct wxFrame wxFrame;
// create
wxFrame *new_wx_frame(const char *title);
// show
void wx_frame_show(wxFrame *frame);
// set menu bar
void wx_frame_set_menu_bar(wxFrame *frame, wxMenuBar *menubar);
//// wxMenuBar
typedef struct wxMenuBar wxMenuBar;
// create
wxMenuBar *new_wx_menu_bar(void);
````

#### src/main.d

`wxWidgets` のウィンドウを作成する D のプログラム。

````d
import std;
import wx;

extern (C) bool wx_init() {
  auto title = "My wxWidgets Window".toStringz;
  auto win = new_wx_frame(title);
  win.wx_frame_show;

  auto menu_bar = new_wx_menu_bar;
  win.wx_frame_set_menu_bar(menu_bar);

  return true;
}

auto main() {

  wx_main(&wx_init);

}
````

#### build.sh

`dub.sdl` を生成して、`dub build` を実行するスクリプト。

````sh
#!/bin/sh -x

## compile C++ source

WX_FLAGS=$(wx-config --cxxflags)
WX_OPT="${WX_FLAGS}"

c++ -c src/wx.cpp -o src/wx.o -std=c++2c ${WX_OPT}

## generate dub.sdl

# name of the compiled file
TARGET_NAME="d_wx_test"

# create an executable
TARGET_TYPE="executable"

# main source files
MAIN_SRC_FILE="src/main.d"

# objects to link
SRC_FILES="src/wx.o"

# system libraries to link
LIBS="stdc++"

# ldc2 linker flags
LFLAGS=$(echo $(for i in $(wx-config --libs); do echo \"${i}\"\ ; done))

# ImportC search path
C_IMPORT_PATH="src"

# dub.sdl
cat <<EOF >dub.sdl

name "${TARGET_NAME}"
targetType "${TARGET_TYPE}"
mainSourceFile "${MAIN_SRC_FILE}"
sourceFiles "${SRC_FILES}"
libs "${LIBS}"
lflags ${LFLAGS}
cImportPaths "${C_IMPORT_PATH}"

EOF

## build

dub clean
dub build
````

#### ビルド

````console
$ chmod +x ./build.sh
$ ./build.sh
````

### D 言語から FLTK を利用する
> date: 2026-05-25  
> topics: d, fltk

`FLTK` の C 言語用のラッパーを作成し、`ImportC` を使用して D 言語のプログラムから利用します。

#### src/fltk.cpp

`FLTK` のラッパー。

````cpp
#include <FL/Fl.H>
#include <FL/Fl_Window.H>

extern "C" {

auto new_fl_window(int left, int top, int width, int height, const char *title)
    -> Fl_Window * {
  auto win = new Fl_Window(left, top, width, height, title);
  return win;
}

void fl_window_show(Fl_Window *win) { win->show(); }

int fl_run() { return Fl::run(); }
}
````

#### src/fltk.h

`FLTK` のラッパーを `ImportC` で使用可能にするためのヘッダー。

````c
typedef struct Fl_Window Fl_Window;

Fl_Window *new_fl_window(int left, int top, int width, int height,
                         const char *title);

void fl_window_show(Fl_Window *win);

int fl_run();
````

#### src/main.d

`FLTK` のウィンドウを作成する D のプログラム。

````d
import std.string : toStringz;

import fltk;

auto main() {

  auto left = 200;
  auto top = 200;
  auto width = 640;
  auto height = 480;
  auto title = "My FLTK Window".toStringz;
  auto win = new_fl_window(left, top, width, height, title);

  fl_window_show(win);

  return fl_run();
}
````

#### dub.sdl

````sdlang
# target name
name "d_fltk_test"

# create an executable
targetType "executable"

# main source files
mainSourceFile "src/main.d"

# objects to link
sourceFiles "src/fltk.o"

# system libraries to link
libs "fltk" "stdc++"

# ldc2 linker flags
lflags "${D_LFLAGS}"

# ImportC search path
cImportPaths "src"
````

#### build.sh
````sh
#!/bin/sh -x

FLTK_INC="-I$(brew --prefix fltk)/include"
FLTK_OPT="${FLTK_INC}"

FLTK_LIB="-L$(brew --prefix fltk)/lib"
export D_LFLAGS=${FLTK_LIB}

c++ src/fltk.cpp -c -o src/fltk.o -std=c++2c ${FLTK_OPT}
dub build
````

#### ビルド
````console

$ ./build.sh
````
