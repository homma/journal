### C++ で SDL3 に含まれる HIDAPI を使用する
> date: 2025-10-29  
> topics: c++, sdl3, hidapi

#### About

C++ のプログラムから SDL3 に含まれている HIDAPI を使用します。  
SDL3 のデフォルトではコントローラのデバイスだけを扱うようになっています。  

- https://wiki.libsdl.org/SDL3/SDL_HINT_HIDAPI_ENUMERATE_ONLY_CONTROLLERS

#### main.cpp

````c++
#include <SDL3/SDL.h>
#include <cwchar>
#include <print>

auto print_dev_info(SDL_hid_device_info *dev_info) -> void {
  std::println("### device ###");
  wprintf(L"manufacturer:     %ls\n", dev_info->manufacturer_string);
  wprintf(L"product:          %ls\n", dev_info->product_string);
  std::println("path:             {}", dev_info->path);
  std::println("vendor id:        {}", dev_info->vendor_id);
  std::println("product id:       {}", dev_info->product_id);
  wprintf(L"serial number:    %ls\n", dev_info->serial_number);
  std::println("release number:   {}", dev_info->release_number);
  std::println("usage page:       {}", dev_info->usage_page);
  std::println("usage:            {}", dev_info->usage);
  std::println("interface number: {}", dev_info->interface_number);

  switch (dev_info->bus_type) {
  case SDL_HID_API_BUS_UNKNOWN:
    std::println("bus type:         {}", "UNKNOWN");
    break;
  case SDL_HID_API_BUS_USB:
    std::println("bus type:         {}", "USB");
    break;
  case SDL_HID_API_BUS_BLUETOOTH:
    std::println("bus type:         {}", "BLUETOOTH");
    break;
  case SDL_HID_API_BUS_I2C:
    std::println("bus type:         {}", "I2C");
    break;
  case SDL_HID_API_BUS_SPI:
    std::println("bus type:         {}", "SPI");
    break;
  }

  std::println();
}

auto print_dev_info(SDL_hid_device *dev) -> void {
  auto dev_info = SDL_hid_get_device_info(dev);
  print_dev_info(dev_info);
}

auto print_all_dev_info() -> void {
  std::println("print devices");
  std::println();

  auto head = SDL_hid_enumerate(0, 0);
  auto dev = head;

  while (true) {
    print_dev_info(dev);

    if (dev->next == NULL) {
      break;
    } else {
      dev = dev->next;
    }
  }

  SDL_hid_free_enumeration(head);
}

auto main() -> int {
  SDL_hid_init();

  print_all_dev_info();

  SDL_hid_exit();
}
````

#### build.sh

````sh
#!/bin/sh -x

SDL3_INC="-I$(brew --prefix sdl3)/include"
SDL3_LIB="-L$(brew --prefix sdl3)/lib -lSDL3"
SDL3_OPT="${SDL3_INC} ${SDL3_LIB}"

CXX_OPT="-std=c++2c"

c++ main.cpp -o test ${CXX_OPT} ${SDL3_OPT}
````

#### References
- https://wiki.libsdl.org/SDL3/CategoryHIDAPI

### Neovim の LSP で clangd を使用する
> date: 2025-10-26  
> topics: c++, neovim

Neovim で C++ のコードを編集しているときに LSP を使うことができるように設定します。

clangd はインストール済み。  
Neovim のバージョンは v0.11.4。  

#### 手順

init.lua ファイルで設定します。  
vim.lsp.config でコンフィグを作成し、vim.lsp.enable で有効にします。

#### init.lua

````lua
-- disable diagnostic
vim.diagnostic.disable()

vim.lsp.config['clangd'] = {
  cmd = { 'clangd' },
  filetypes = { 'c', 'cpp' },
}
vim.lsp.enable('clangd')
````

#### References
- https://neovim.io/doc/user/lsp.html

### raygui で日本語を表示する
> date: 2025-10-17  
> topics: c++, raylib

#### About

raygui で日本語を表示します。  
フォントは `Noto Sans Japanese` を使用します。

#### raygui のダウンロード

````console
$ git clone --depth 1 https://github.com/raysan5/raygui.git
````

#### Code

main.cpp
````c++
#include <print>
#include <raylib.h>

#define RAYGUI_IMPLEMENTATION
#include <raygui.h>

namespace font {

Font font;
auto path = "./static/NotoSansJP-Regular.ttf";
auto glyph_count = 65535;
auto size = 24;

} // namespace font

auto init_ui() -> void {
  font::font = LoadFontEx(font::path, font::size, 0, font::glyph_count);
  GuiSetFont(font::font);
  GuiSetStyle(DEFAULT, TEXT_SIZE, font::size);
}

auto deinit_ui() -> void { UnloadFont(font::font); }

auto draw_ui() -> void {
  auto bg_color = GetColor(GuiGetStyle(DEFAULT, BACKGROUND_COLOR));
  ClearBackground(bg_color);

  if (GuiButton(Rectangle{200, 160, 200, 80}, "ボタン")) {
    std::println("ボタンがクリックされました");
  }
}

auto update() -> void {
  BeginDrawing();

  draw_ui();

  EndDrawing();
}

auto main() -> int {
  InitWindow(600, 400, "Using Custom Font");

  init_ui();

  while (!WindowShouldClose()) {
    update();
  }

  deinit_ui();

  CloseWindow();
}
````

build.sh
````sh
#!/bin/sh -x

RAYLIB_INC="-I$(brew --prefix raylib)/include"
RAYLIB_LIB="-L$(brew --prefix raylib)/lib"
RAYLIB_OPT="${RAYLIB_INC} ${RAYLIB_LIB} -lraylib"
RAYGUI_OPT="-Iraygui/src"

c++ main.cpp -o test_font -std=c++2c ${RAYLIB_OPT} ${RAYGUI_OPT}
````

#### ビルドと実行

````console
$ vi main.cpp
$ vi build.sh
$ chmod +x ./build.sh
$ ./build.sh
$ ./test_font
````

### raylib で日本語を表示する
> date: 2025-10-14  
> topics: c++, raylib

#### About

raylib で日本語を表示します。  
フォントは `Noto Sans Japanese` を使用します。

#### Code

main.cpp
````c++
#include <raylib.h>

auto txt = "こんにちは、日本語";

Font fnt;
auto font_path = "./static/NotoSansJP-Regular.ttf";
// Noto Sans Japanese contains 65535 glyphs as described in
// https://fonts.google.com/noto/specimen/Noto+Sans+JP/about
auto glyph_count = 65535;
auto font_size = 24;

auto update() -> void {
  BeginDrawing();

  ClearBackground(WHITE);

  auto position = Vector2(100, 100);
  auto spacing = 1.0f;

  DrawTextEx(fnt, txt, position, font_size, spacing, BLACK);

  EndDrawing();
}

auto main() -> int {
  InitWindow(600, 400, "Using Custom Font");

  fnt = LoadFontEx(font_path, font_size, 0, glyph_count);

  while (!WindowShouldClose()) {
    update();
  }

  UnloadFont(fnt);

  CloseWindow();
}
````

### Cpp2 で SDL3 のウィンドウを表示する
> date: 2025-10-04  
> topics: c++, cpp2, sdl3

#### About

Cpp2 から SDL3 を使用し、ウィンドウを表示します。

#### Code

main.cpp2
````cpp2
#include <SDL3/SDL.h>

main: () = {
  if !SDL_Init(SDL_INIT_VIDEO) {
    std::println("error");
    std::exit(1);
  }

  win: * SDL_Window;
  {
    title := "My SDL3 Window";
    w := 600;
    h := 400;
    flags := 0;

    win = SDL_CreateWindow(title, w, h, flags);
  }

  event := SDL_Event();
  quit := false;

  while quit != true {
    while SDL_PollEvent(event&) {
      if event.type == SDL_EVENT_QUIT {
        quit = true;
      }
    }
  }

  SDL_DestroyWindow(win);
  SDL_Quit();
}
````

> date: 2025-10-04

#### About

Cpp2 をインストールし、実際に使ってみます。

#### Hello, World

hello.cpp2

````cpp2
  std::println("Hello, World");
}
````

build.sh
````sh
#!/bin/sh -x

PATH=./cppfront/source:${PATH}
CPPFRONT_INCLUDE=./cppfront/include
FILENAME=$(echo ${1} | cut -f 1 -d .)

cppfront ${1} -cl -in -p -o ${FILENAME}.cpp
c++ ${FILENAME}.cpp -std=c++2c -I${CPPFRONT_INCLUDE} -o ${FILENAME}
````

ビルドと実行
````console
$ vi hello.cpp2
$ vi build.sh
$ chmod +x ./build.sh
$ ./build.sh hello.cpp2
$ ./hello 
Hello, World
````

#### cppfront コマンドのオプション

- `-vers` : バージョンを出力する
- `-h` : ヘルプを出力する
- `-p` : Cpp2 の文法のみを受け入れる（C++ の文法と混ぜられない）
- `-in` : `std` ヘッダーを全てインクルードする
- `-cl` : メタ情報を出力しない
- `-o <filename>` : 出力先のファイル名を指定

#### メモ
C++ の `import` について
- `Apple clang version 17.0.0 (clang-1700.3.19.1)` では C++ の `import` が使えない
- 代わりに `-in` オプションで `std` ヘッダーをインクルードする

シンタックスハイライトについて
- Neovim では `.cpp2` ファイルのシンタックスハイライトが効かない
- GitHub も `cpp2` のシンタックスハイライトはできなさそう

コードフォーマッターについて
- Cpp2 のフォーマッターは標準装備されていなさそう
- AST を出力するためのルーチンは実装されている
- AST が出力できれば、AST を元にフォーマットできるかもしれない
