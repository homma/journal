### raygui で日本語を表示する
> date: 2025-10-17  
> topics: c++, raylib

#### About

raygui で日本語を表示します。  
フォントは `Noto Sans Japanese` を使用します。

#### raygui のダウンロード

````console
$ git clone --depth 1 https://github.com/raysan5/raygui.git
````

#### Code

main.cpp
````c++
#include <print>
#include <raylib.h>

#define RAYGUI_IMPLEMENTATION
#include <raygui.h>

namespace font {

Font font;
auto path = "./static/NotoSansJP-Regular.ttf";
auto glyph_count = 65535;
auto size = 24;

} // namespace font

auto init_ui() -> void {
  font::font = LoadFontEx(font::path, font::size, 0, font::glyph_count);
  GuiSetFont(font::font);
  GuiSetStyle(DEFAULT, TEXT_SIZE, font::size);
}

auto deinit_ui() -> void { UnloadFont(font::font); }

auto draw_ui() -> void {
  auto bg_color = GetColor(GuiGetStyle(DEFAULT, BACKGROUND_COLOR));
  ClearBackground(bg_color);

  if (GuiButton(Rectangle{200, 160, 200, 80}, "ボタン")) {
    std::println("ボタンがクリックされました");
  }
}

auto update() -> void {
  BeginDrawing();

  draw_ui();

  EndDrawing();
}

auto main() -> int {
  InitWindow(600, 400, "Using Custom Font");

  init_ui();

  while (!WindowShouldClose()) {
    update();
  }

  deinit_ui();

  CloseWindow();
}
````

build.sh
````sh
#!/bin/sh -x

RAYLIB_INC="-I$(brew --prefix raylib)/include"
RAYLIB_LIB="-L$(brew --prefix raylib)/lib"
RAYLIB_OPT="${RAYLIB_INC} ${RAYLIB_LIB} -lraylib"
RAYGUI_OPT="-Iraygui/src"

c++ main.cpp -o test_font -std=c++2c ${RAYLIB_OPT} ${RAYGUI_OPT}
````

#### ビルドと実行

````console
$ vi main.cpp
$ vi build.sh
$ chmod +x ./build.sh
$ ./build.sh
$ ./test_font
````

### raylib で日本語を表示する
> date: 2025-10-14  
> topics: c++, raylib

#### About

raylib で日本語を表示します。  
フォントは `Noto Sans Japanese` を使用します。

#### Code

main.cpp
````c++
#include <raylib.h>

auto txt = "こんにちは、日本語";

Font fnt;
auto font_path = "./static/NotoSansJP-Regular.ttf";
// Noto Sans Japanese contains 65535 glyphs as described in
// https://fonts.google.com/noto/specimen/Noto+Sans+JP/about
auto glyph_count = 65535;
auto font_size = 24;

auto update() -> void {
  BeginDrawing();

  ClearBackground(WHITE);

  auto position = Vector2(100, 100);
  auto spacing = 1.0f;

  DrawTextEx(fnt, txt, position, font_size, spacing, BLACK);

  EndDrawing();
}

auto main() -> int {
  InitWindow(600, 400, "Using Custom Font");

  fnt = LoadFontEx(font_path, font_size, 0, glyph_count);

  while (!WindowShouldClose()) {
    update();
  }

  UnloadFont(fnt);

  CloseWindow();
}
````

````sh
#!/bin/sh -x

RAYLIB_INC="-I$(brew --prefix raylib)/include"
RAYLIB_LIB="-L$(brew --prefix raylib)/lib"
RAYLIB_OPT="${RAYLIB_INC} ${RAYLIB_LIB} -lraylib"

c++ main.cpp -o test_font -std=c++2c ${RAYLIB_OPT}
````

### Cpp2 で raylib を使用する
> date: 2025-10-04  
> topics: c++, cpp2, raylib

#### About

Cpp2 から raylib を使用してみます。

#### Code

main.cpp2
````cpp2
#include <raylib.h>

white: Color = (255, 255, 255, 255);
black: Color = (0, 0, 0, 255);

update: () = {

  BeginDrawing();

  ClearBackground(white);

  {
    text := "This is a raylib window";
    x := 110;
    y := 170;
    h := 30;
    color := black;

    DrawText(text, x, y, h, color);
  }

  EndDrawing();
}

main: () = {
  w := 600;
  h := 400;
  title := "My Raylib Window";

  InitWindow(w, h, title);

  while !WindowShouldClose() {
    update();
  }

  CloseWindow();
}
````

build.sh
````sh
#!/bin/sh -x

PATH=./cppfront/source:${PATH}
CPPFRONT_OPT="-I./cppfront/include"

RAYLIB_INC="-I$(brew --prefix raylib)/include"
RAYLIB_LIB="-L$(brew --prefix raylib)/lib -lraylib"
RAYLIB_OPT="${RAYLIB_INC} ${RAYLIB_LIB}"

FILENAME=$(echo ${1} | cut -f 1 -d .)

cppfront ${1} -cl -in -o ${FILENAME}.cpp
c++ ${FILENAME}.cpp -std=c++2c ${RAYLIB_OPT} ${CPPFRONT_OPT} -o ${FILENAME}
````

#### ビルドと実行


### Cpp2 で SDL3 のウィンドウを表示する
> date: 2025-10-04  
> topics: c++, cpp2, sdl3

#### About

Cpp2 から SDL3 を使用し、ウィンドウを表示します。

#### Code

main.cpp2
````cpp2
#include <SDL3/SDL.h>

main: () = {
  if !SDL_Init(SDL_INIT_VIDEO) {
    std::println("error");
    std::exit(1);
  }

  win: * SDL_Window;
  {
    title := "My SDL3 Window";
    w := 600;
    h := 400;
    flags := 0;

    win = SDL_CreateWindow(title, w, h, flags);
  }

  event := SDL_Event();
  quit := false;

  while quit != true {
    while SDL_PollEvent(event&) {
      if event.type == SDL_EVENT_QUIT {
        quit = true;
      }
    }
  }

  SDL_DestroyWindow(win);
  SDL_Quit();
}
````


### Cpp2 を試す
> date: 2025-10-04  
> topics: c++, cpp2

#### About

Cpp2 をインストールし、実際に使ってみます。

#### インストール

````console
$ git clone --depth 1 https://github.com/hsutter/cppfront.git
$ cd cppfront/source
$ c++ cppfront.cpp -std=c++20 -o cppfront
$ ./cppfront -vers

SPDX-License-Identifier  Apache-2.0 WITH LLVM-exception
Copyright (c) 2022-2025  Herb Sutter
````

#### Hello, World

hello.cpp2
````cpp2
main: () = {
  std::println("Hello, World");
}
````

build.sh
````sh
#!/bin/sh -x

PATH=./cppfront/source:${PATH}
CPPFRONT_INCLUDE=./cppfront/include
FILENAME=$(echo ${1} | cut -f 1 -d .)

cppfront ${1} -cl -in -p -o ${FILENAME}.cpp
c++ ${FILENAME}.cpp -std=c++2c -I${CPPFRONT_INCLUDE} -o ${FILENAME}
````

ビルドと実行
````console
$ vi hello.cpp2
$ vi build.sh
$ chmod +x ./build.sh
$ ./build.sh hello.cpp2
$ ./hello 
Hello, World
````

#### cppfront コマンドのオプション

- `-vers` : バージョンを出力する
- `-h` : ヘルプを出力する
- `-p` : Cpp2 の文法のみを受け入れる（C++ の文法と混ぜられない）
- `-in` : `std` ヘッダーを全てインクルードする
- `-cl` : メタ情報を出力しない
- `-o <filename>` : 出力先のファイル名を指定

#### メモ
C++ の `import` について
- `Apple clang version 17.0.0 (clang-1700.3.19.1)` では C++ の `import` が使えない
- 代わりに `-in` オプションで `std` ヘッダーをインクルードする

シンタックスハイライトについて
- Neovim では `.cpp2` ファイルのシンタックスハイライトが効かない
- GitHub も `cpp2` のシンタックスハイライトはできなさそう

コードフォーマッターについて
- Cpp2 のフォーマッターは標準装備されていなさそう
- AST を出力するためのルーチンは実装されている
- AST が出力できれば、AST を元にフォーマットできるかもしれない
