### Objective-C++ で音声ファイルを再生する
> date: 2025-11-03  
> topics: c++, objective-c++, audio  

#### About

Objective-C++ を使用して音声ファイルを再生します。  

#### main.mm

````mm
#import <AVFAudio/AVAudioPlayer.h>
#include <chrono>
#include <print>
#include <thread>

auto main(int argc, char *argv[]) -> int {
  if (argc < 2) {
    std::println("Usage: play <file>");
    exit(1);
  }

  NSString *path = [NSString stringWithUTF8String:argv[1]];
  NSURL *url = [NSURL fileURLWithPath:path];
  NSError *error;
  AVAudioPlayer *player = [[AVAudioPlayer alloc] initWithContentsOfURL:url
                                                                 error:&error];
  if (player == nil) {
    NSLog(@"cannot create audio player: %@", error);
    exit(1);
  }

  [player play];

  while ([player isPlaying]) {
    auto duration = std::chrono::milliseconds(100);
    std::this_thread::sleep_for(duration);
  }
}
````

#### build.sh

````sh
#!/bin/sh -x

AVFAUDIO_OPT="-framework AVFAudio"
OBJC_OPT="-framework Foundation"
CXX_OPT="-ObjC++ --std=c++2b"

c++ main.mm -o play ${CXX_OPT} ${OBJC_OPT} ${AVFAUDIO_OPT}
````
