### Raylib で音声ファイルを再生する
> date: 2025-11-24  
> topics: c++, raylib, audio  

#### About

Raylib を使用して音声ファイルを再生します。  

#### main.cpp

````cpp
#include <filesystem>
#include <print>
#include <raylib.h>

std::optional<Sound> sound1;
std::optional<Sound> sound2;

auto open_file(std::string file) -> std::optional<Sound> {

  auto path = std::filesystem::path{file};
  auto exists = std::filesystem::exists(path);

  if (not exists) {
    std::println("Error: cannot open {}", file);
    return std::nullopt;
  }

  auto sound = LoadSound(file.c_str());

  return sound;
}

auto close_file(std::optional<Sound> sound) -> void {
  if (not sound) {
    return;
  }

  UnloadSound(sound.value());
}

auto play_sound(std::optional<Sound> sound) -> void {
  if (not sound) {
    return;
  }

  PlaySound(sound.value());
}

auto update() -> void {

  if (IsKeyPressed(KEY_J)) {
    play_sound(sound1);
  }

  if (IsKeyPressed(KEY_K)) {
    play_sound(sound2);
  }

  BeginDrawing();

  ClearBackground(WHITE);

  {
    auto text = "Press J to play assets/sound1.mp3";
    auto left = 100;
    auto top = 100;
    auto font_size = 20;
    DrawText(text, left, top, font_size, GRAY);
  }

  {
    auto text = "Press K to play assets/sound2.mp3";
    auto left = 100;
    auto top = 130;
    auto font_size = 20;
    DrawText(text, left, top, font_size, GRAY);
  }

  EndDrawing();
}

auto main() -> int {
  SetTargetFPS(60);
  InitWindow(600, 400, "Playing Sounds");
  InitAudioDevice();

  sound1 = open_file("assets/sound1.mp3");
  sound2 = open_file("assets/sound2.mp3");

  while (!WindowShouldClose()) {
    update();
  }

  close_file(sound1);
  close_file(sound2);

  CloseAudioDevice();
  CloseWindow();
}
````

#### build.sh

````sh
#!/bin/sh -x

RAYLIB_INC="-I$(brew --prefix raylib)/include"
RAYLIB_LIB="-L$(brew --prefix raylib)/lib"
RAYLIB_OPT="${RAYLIB_INC} ${RAYLIB_LIB} -lraylib"

c++ main.cpp -o test_music -std=c++2c ${RAYLIB_OPT}
````

#### 実行

- `assets/sound1.mp3` と `assets/sound2.mp3` ファイルを用意します
- `./test_music` コマンドでプログラムを実行します
- `j` キーを押すと `assets/sound1.mp3` ファイルが再生されます
- `k` キーを押すと `assets/sound2.mp3` ファイルが再生されます

### Objective-C++ で音声ファイルを再生する
> date: 2025-11-03  
> topics: c++, objective-c++, audio  

#### About

Objective-C++ を使用して音声ファイルを再生します。  

#### main.mm

````mm
#import <AVFAudio/AVAudioPlayer.h>
#include <chrono>
#include <print>
#include <thread>

auto main(int argc, char *argv[]) -> int {
  if (argc < 2) {
    std::println("Usage: play <file>");
    exit(1);
  }

  NSString *path = [NSString stringWithUTF8String:argv[1]];
  NSURL *url = [NSURL fileURLWithPath:path];
  NSError *error;
  AVAudioPlayer *player = [[AVAudioPlayer alloc] initWithContentsOfURL:url
                                                                 error:&error];
  if (player == nil) {
    NSLog(@"cannot create audio player: %@", error);
    exit(1);
  }

  [player play];

  while ([player isPlaying]) {
    auto duration = std::chrono::milliseconds(100);
    std::this_thread::sleep_for(duration);
  }
}
````

#### build.sh

````sh
#!/bin/sh -x

AVFAUDIO_OPT="-framework AVFAudio"
OBJC_OPT="-framework Foundation"
CXX_OPT="-ObjC++ --std=c++2b"

c++ main.mm -o play ${CXX_OPT} ${OBJC_OPT} ${AVFAUDIO_OPT}
````
