### C++ から SDL3 を使用する
> date: 2025-04-13  
> topics: c++, sdl3  

#### SDL3 のインストール

````console
$ brew install sdl3
````

#### SConstruct
````python
import subprocess

# Executable
target_name = "sdl_test"
src = "main.cpp"

# Library
## SDL3
sdl3 = {}
sdl3["lib"] = "SDL3"
sdl3["path"] = (
    subprocess.Popen(["brew", "--prefix", "sdl3"], stdout=subprocess.PIPE)
    .stdout.read()
    .decode("UTF-8")
    .strip()
)
sdl3["lib_path"] = sdl3["path"] + "/lib"
sdl3["include_path"] = sdl3["path"] + "/include"

Program(
    target_name,
    src,
    LIBS=[sdl3["lib"]],
    LIBPATH=[sdl3["lib_path"]],
    CPPPATH=[sdl3["include_path"]],
    CCFLAGS=["-std=c++2c"],
)
````

#### main.cpp
````cpp
#include <SDL3/SDL.h>
#include <print>

auto main() -> int {
  auto v = SDL_GetVersion();
  std::println("{}.{}.{}", SDL_VERSIONNUM_MAJOR(v), SDL_VERSIONNUM_MINOR(v),
               SDL_VERSIONNUM_MICRO(v));
}
````

#### ビルドと実行
````console
$ scons
$ ./sdl_test
3.2.10
````
